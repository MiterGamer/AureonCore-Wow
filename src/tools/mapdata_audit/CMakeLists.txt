add_executable(mapdata_audit main.cpp)
target_link_libraries(mapdata_audit PRIVATE trinity-core-interface scripts game)
target_include_directories(mapdata_audit PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
set(store_file "${CMAKE_SOURCE_DIR}/src/server/game/DataStores/DB2Stores.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${store_file}")
file(STRINGS "${store_file}" store_lines REGEX "DB2Storage<")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/tables.inc" "// Generated from the server's DB2 storage declarations.\n")
foreach(store_line IN LISTS store_lines)
  if(store_line MATCHES "\\(\"([^\"]+\\.db2)\", &([A-Za-z0-9_]+)::Instance\\)")
    file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/tables.inc" "{ \"${CMAKE_MATCH_1}\", &${CMAKE_MATCH_2}::Instance },\n")
  endif()
endforeach()
set_target_properties(mapdata_audit PROPERTIES FOLDER "tools")
